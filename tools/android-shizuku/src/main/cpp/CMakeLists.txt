cmake_minimum_required(VERSION 3.22.1)
project(operit_shizuku LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
add_compile_options(-fno-exceptions -fno-rtti -fno-threadsafe-statics)
find_package(boringssl REQUIRED CONFIG)
find_package(cxx REQUIRED CONFIG)
set(UPSTREAM "${CMAKE_CURRENT_LIST_DIR}/../../../upstream")
add_library(operit_adb SHARED "${UPSTREAM}/manager-jni/adb_pairing.cpp")
target_link_libraries(operit_adb log boringssl::crypto_static cxx::cxx)
add_executable(liboperit_shizuku.so starter.cpp)
target_link_libraries(liboperit_shizuku.so log)
